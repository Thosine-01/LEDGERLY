making migrations
# 1. Confirm .venv is active
which python                      # must show .venv path

# 2. Confirm Postgres is running
docker ps | grep ledgerly-postgres  # must show (healthy)

# 3. Generate the migration
python manage.py makemigrations merchant

# 4. Apply it
python manage.py migrate

# 5. Create a superuser
python manage.py createsuperuser

# 6. Verify tables in Postgres
docker exec -it ledgerly-postgres psql -U ledgerly -d ledgerly_dev -c "\dt";
